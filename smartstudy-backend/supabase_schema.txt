-- SmartStudy / PAATAM.AI Supabase schema

create extension if not exists vector;
create extension if not exists pgcrypto;

create table if not exists public.subjects (
  id text primary key,
  name text not null unique,
  created_at timestamptz not null default now()
);

-- A teacher's class, e.g. "Grade 10 - A". Top level of the Classes navigation.
create table if not exists public.classes (
  id text primary key,
  name text not null,
  section text,
  created_at timestamptz not null default now()
);

-- Section is mandatory going forward for new classes.
alter table public.classes add column if not exists section text;

-- A subject taught within a class, e.g. "Physics" inside "Grade 10 - A".
create table if not exists public.class_subjects (
  id text primary key,
  class_id text not null references public.classes(id) on delete cascade,
  subject_name text not null,
  created_at timestamptz not null default now(),
  unique (class_id, subject_name)
);

create table if not exists public.assignments (
  id text primary key,
  title text not null,
  class_name text not null,
  subject text,
  language text not null default 'English',
  difficulty text not null default 'Medium',
  duration_minutes integer not null default 60 check (duration_minutes > 0),
  questions_json jsonb not null default '[]'::jsonb,
  reasoning_model text,
  status text not null default 'dispatched',
  created_at timestamptz not null default now()
);

alter table public.assignments add column if not exists subject text;
alter table public.assignments add column if not exists language text not null default 'English';
alter table public.assignments add column if not exists difficulty text not null default 'Medium';
alter table public.assignments add column if not exists duration_minutes integer not null default 60;
alter table public.assignments add column if not exists reasoning_model text;
-- Question paper deployment scoping: which class/subject this paper belongs to.
alter table public.assignments add column if not exists class_id text references public.classes(id) on delete set null;
alter table public.assignments add column if not exists class_subject_id text references public.class_subjects(id) on delete set null;
-- Canonical teacher-supplied answer key/marking scheme for this paper, used by grading.
alter table public.assignments add column if not exists answer_key_text text;
-- Original source documents used by the teacher review workspace.
alter table public.assignments add column if not exists question_paper_url text;
alter table public.assignments add column if not exists answer_key_url text;

create index if not exists class_subjects_class_id_idx on public.class_subjects (class_id);
create index if not exists assignments_class_subject_id_idx on public.assignments (class_subject_id);

create table if not exists public.submissions (
  id text primary key,
  assignment_id text references public.assignments(id) on delete set null,
  student_name text not null,
  subject text,
  language text not null default 'English',
  lang_code text,
  sample_paper_url text,
  sample_paper_urls jsonb not null default '[]'::jsonb,
  ocr_text text,
  score numeric,
  max_score numeric,
  feedback text,
  socratic_hint text,
  ai_evaluation_json jsonb,
  evaluation_provider text,
  evaluation_status text,
  fallback_reason text,
  status text not null default 'pending_review',
  submitted_at timestamptz not null default now(),
  final_score numeric,
  final_feedback text,
  final_hint text,
  final_evaluation_json jsonb
);

alter table public.submissions add column if not exists sample_paper_urls jsonb not null default '[]'::jsonb;
alter table public.submissions add column if not exists ai_evaluation_json jsonb;
alter table public.submissions add column if not exists final_score numeric;
alter table public.submissions add column if not exists max_score numeric;
alter table public.submissions alter column score type numeric using score::numeric;
alter table public.submissions alter column final_score type numeric using final_score::numeric;
alter table public.submissions add column if not exists final_feedback text;
alter table public.submissions add column if not exists final_hint text;
alter table public.submissions add column if not exists evaluation_provider text;
alter table public.submissions add column if not exists evaluation_status text;
alter table public.submissions add column if not exists fallback_reason text;
alter table public.submissions add column if not exists final_evaluation_json jsonb;

create table if not exists public.parents (
  id text primary key default gen_random_uuid()::text,
  name text not null,
  phone_number text not null unique,
  email text,
  preferred_channel text default 'whatsapp',
  created_at timestamptz not null default now()
);

create table if not exists public.students (
  id text primary key default gen_random_uuid()::text,
  name text not null,
  roll_number text,
  class_id text references public.classes(id) on delete set null,
  parent_id text references public.parents(id) on delete set null,
  created_at timestamptz not null default now()
);

alter table public.submissions add column if not exists student_id text references public.students(id) on delete set null;

create table if not exists public.notifications (
  id text primary key,
  type text not null,
  title text not null,
  message text not null,
  details_json jsonb,
  timestamp timestamptz not null default now(),
  student_name text,
  student_id text references public.students(id) on delete set null,
  parent_phone text,
  delivery_status text default 'pending',
  provider_message_id text
);

create table if not exists public.textbook_embeddings (
  id uuid primary key default gen_random_uuid(),
  class_name text not null,
  chapter_title text not null,
  content_chunk text not null,
  embedding vector(1536) not null,
  created_at timestamptz not null default now()
);

create index if not exists assignments_created_at_idx on public.assignments (created_at desc);
create index if not exists submissions_submitted_at_idx on public.submissions (submitted_at desc);
create index if not exists submissions_assignment_id_idx on public.submissions (assignment_id);
create index if not exists textbook_embeddings_class_name_idx on public.textbook_embeddings (class_name);

drop function if exists public.match_textbook_chunks(vector, double precision, integer, text);

create function public.match_textbook_chunks(
  query_embedding vector(1536),
  match_threshold double precision,
  match_count integer,
  filter_class_name text
)
returns table (
  id uuid,
  class_name text,
  chapter_title text,
  content_chunk text,
  similarity double precision
)
language sql
stable
as $$
  select
    textbook_embeddings.id,
    textbook_embeddings.class_name,
    textbook_embeddings.chapter_title,
    textbook_embeddings.content_chunk,
    (1 - (textbook_embeddings.embedding <=> query_embedding))::double precision as similarity
  from public.textbook_embeddings
  where textbook_embeddings.class_name = filter_class_name
    and 1 - (textbook_embeddings.embedding <=> query_embedding) >= match_threshold
  order by textbook_embeddings.embedding <=> query_embedding
  limit greatest(match_count, 0);
$$;

grant select, insert, update, delete on public.class_subjects to anon, authenticated, service_role;
grant select, insert, update, delete on public.assignments to anon, authenticated, service_role;
grant select, insert, update, delete on public.submissions to anon, authenticated, service_role;
grant select, insert, update, delete on public.notifications to anon, authenticated, service_role;
grant select, insert, update, delete on public.textbook_embeddings to anon, authenticated, service_role;
grant execute on function public.match_textbook_chunks(vector, double precision, integer, text) to anon, authenticated, service_role;

insert into storage.buckets (id, name, public)
values ('student-submissions', 'student-submissions', true)
on conflict (id) do update set public = true;

do $$
begin
  if not exists (select 1 from pg_policies where schemaname = 'storage' and tablename = 'objects' and policyname = 'SmartStudy public read') then
    create policy "SmartStudy public read" on storage.objects for select using (bucket_id = 'student-submissions');
  end if;
  if not exists (select 1 from pg_policies where schemaname = 'storage' and tablename = 'objects' and policyname = 'SmartStudy upload') then
    create policy "SmartStudy upload" on storage.objects for insert to anon, authenticated with check (bucket_id = 'student-submissions');
  end if;
  if not exists (select 1 from pg_policies where schemaname = 'storage' and tablename = 'objects' and policyname = 'SmartStudy update') then
    create policy "SmartStudy update" on storage.objects for update to anon, authenticated using (bucket_id = 'student-submissions') with check (bucket_id = 'student-submissions');
  end if;
end $$;
